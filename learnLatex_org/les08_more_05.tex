% Lesson 08: Tables: More
% Manipulating the space between columns: Arbitrary separation with ! (it adds its argument in center of the space between two columns.)
\documentclass{article}
\usepackage[T1]{fontenc}
\usepackage{array}

\begin{document}
	\begin{tabular}{l !{:} ll}
		Animal & Food  & Size   \\
		dog    & meat  & medium \\
		horse  & hay   & large  \\
		frog   & flies & small  \\
	\end{tabular}
\end{document}
